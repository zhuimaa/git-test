class _StarRocksClient:
    """极简 MySQL 协议客户端，仅标准库 socket/struct/hashlib，无需任何第三方依赖"""

    def __init__(self, host, port, user, password, database, connect_timeout=10):
        import socket
        self._sock = socket.create_connection((host, port), timeout=connect_timeout)
        self._seq = 0
        self._user = user
        self._password = password
        self._database = database
        self._handshake()

    # ---- 底层 IO ----
    def _recv_exact(self, n):
        data = b""
        while len(data) < n:
            chunk = self._sock.recv(n - len(data))
            if not chunk:
                raise ConnectionError("StarRocks 连接被关闭")
            data += chunk
        return data

    def _read_packet(self):
        header = self._recv_exact(4)
        import struct
        length = struct.unpack("<I", header[:3] + b"\x00")[0]
        seq = header[3]
        payload = self._recv_exact(length) if length > 0 else b""
        self._seq = (seq + 1) % 256
        return payload

    def _send_packet(self, payload):
        import struct
        header = struct.pack("<I", len(payload))[:3] + bytes([self._seq])
        self._sock.sendall(header + payload)
        self._seq = (self._seq + 1) % 256

    # ---- 协议解析 ----
    @staticmethod
    def _read_lenenc(data, offset):
        first = data[offset]
        import struct
        if first < 0xFB:
            return first, offset + 1
        if first == 0xFC:
            return struct.unpack("<H", data[offset + 1:offset + 3])[0], offset + 3
        if first == 0xFD:
            return struct.unpack("<I", data[offset + 1:offset + 4] + b"\x00")[0], offset + 4
        if first == 0xFE:
            return struct.unpack("<Q", data[offset + 1:offset + 9])[0], offset + 9
        return 0, offset + 1  # 0xFB NULL 当做 0 长度

    @staticmethod
    def _skip_lenenc(data, offset):
        first = data[offset]
        if first < 0xFB:
            return offset + 1
        if first == 0xFC:
            return offset + 3
        if first == 0xFD:
            return offset + 4
        if first == 0xFE:
            return offset + 9
        return offset + 1

    # ---- 握手认证 ----
    def _handshake(self):
        import struct, hashlib

        payload = self._read_packet()
        pos = payload.index(b"\x00", 1) + 1  # 跳过 protocol_version + server_version
        pos += 4                               # connection_id
        auth1 = payload[pos:pos + 8]; pos += 8
        pos += 1                               # filler
        pos += 2                               # cap_lower (忽略)
        pos += 1                               # charset (忽略)
        pos += 2                               # status (忽略)
        pos += 2                               # cap_upper (忽略)
        auth_data_len = payload[pos]; pos += 1
        pos += 10                              # reserved
        auth2_len = max(auth_data_len - 8, 12) if auth_data_len > 8 else 12
        auth2 = payload[pos:pos + auth2_len]
        scramble = auth1 + auth2

        password = self._password.encode()
        hash1 = hashlib.sha1(password).digest()
        hash2 = hashlib.sha1(hash1).digest()
        hash3 = hashlib.sha1(scramble + hash2).digest()
        auth_resp = bytes(a ^ b for a, b in zip(hash1, hash3))

        # 构建握手响应
        CLIENT_PROTOCOL_41 = 0x200
        CLIENT_SECURE_CONNECTION = 0x8000
        CLIENT_PLUGIN_AUTH = 0x80000
        CLIENT_CONNECT_WITH_DB = 0x8
        caps = (CLIENT_PROTOCOL_41 | CLIENT_SECURE_CONNECTION |
                CLIENT_PLUGIN_AUTH | CLIENT_CONNECT_WITH_DB)

        resp = bytearray()
        resp += struct.pack("<I", caps)
        resp += struct.pack("<I", 16777215)           # max_packet_size
        resp += bytes([45])                            # utf8mb4
        resp += b"\x00" * 23
        resp += self._user.encode() + b"\x00"
        resp += bytes([len(auth_resp)]) + auth_resp
        resp += self._database.encode() + b"\x00"
        resp += b"mysql_native_password\x00"

        self._send_packet(bytes(resp))
        ok = self._read_packet()
        if ok[0] == 0xFF:
            code = struct.unpack("<H", ok[1:3])[0]
            msg = ok[3:].decode("utf8", errors="replace")
            raise ConnectionError(f"StarRocks 认证失败 [{code}]: {msg}")

    # ---- 查询 ----
    def query(self, sql):
        import struct
        self._send_packet(bytes([0x03]) + sql.encode())

        # 列数
        pkt = self._read_packet()
        if pkt[0] == 0xFF:
            code = struct.unpack("<H", pkt[1:3])[0]
            msg = pkt[3:].decode("utf8", errors="replace")
            raise RuntimeError(f"查询失败 [{code}]: {msg}")

        col_count, _ = self._read_lenenc(pkt, 0)

        # 列定义 — 只取列名
        columns = []
        for _ in range(col_count):
            cp = self._read_packet()
            pos = 0
            for _ in range(4):              # catalog, schema, table, org_table
                pos = self._skip_lenenc(cp, pos)
            nlen, pos = self._read_lenenc(cp, pos)
            columns.append(cp[pos:pos + nlen].decode("utf8"))
            pos += nlen

        # EOF (0xFE, len<9) 或没有 (DEPRECATE_EOF)
        # 用"再读一包，如果像是数据行则直接处理"的策略来兼容两种模式
        maybe_eof = self._read_packet()
        if maybe_eof[0] == 0xFE and len(maybe_eof) < 9:
            pass  # 经典 EOF，丢弃
        else:
            # 这不是 EOF，是第一条数据行，回退处理
            self._row_buffer = maybe_eof
            self._row_buffer_idx = 0

        # 读数据行
        rows = []
        col_count = len(columns)
        while True:
            # 优先取缓冲
            if hasattr(self, "_row_buffer") and self._row_buffer is not None:
                rp = self._row_buffer
                self._row_buffer = None
            else:
                rp = self._read_packet()

            # EOF 包 (0xFE, len<9) 或 OK 包 (0xFE 或 0x00, 需看上下文)
            if rp[0] == 0xFE and len(rp) < 9:
                break
            if rp[0] == 0x00 and len(rp) < 9:
                # 检查是否 OK 包 (affected_rows=0, insert_id=0)
                # OK 包格式: 0x00 + lenenc(0) + lenenc(0) + 2status + 2warnings
                if len(rp) >= 7:
                    break

            pos = 0
            row = {}
            for ci in range(col_count):
                if rp[pos] == 0xFB:
                    row[columns[ci]] = None
                    pos += 1
                else:
                    vlen, pos = self._read_lenenc(rp, pos)
                    row[columns[ci]] = rp[pos:pos + vlen].decode("utf8")
                    pos += vlen
            rows.append(row)

        return columns, rows

    def close(self):
        try:
            self._send_packet(bytes([0x01]))  # COM_QUIT
            self._sock.close()
        except Exception:
            pass

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()


def get_starrocks_connection():
    """
    使用纯标准库 MySQL 协议连接 StarRocks

    返回值：
        _StarRocksClient 对象

    异常：
        连接失败时返回 None，切换到文件存储模式
    """
    global USE_FILE_STORAGE
    try:
        return _StarRocksClient(
            host=STARROCKS_HOST, port=STARROCKS_PORT,
            user=STARROCKS_USER, password=STARROCKS_PASSWORD,
            database=STARROCKS_DATABASE, connect_timeout=10,
        )
    except Exception as e:
        print(f"StarRocks 连接失败: {e}，切换到文件存储模式")
        USE_FILE_STORAGE = True
        return None


def execute_query(sql: str, params: list = None) -> tuple:
    """
    执行 SELECT 查询（纯标准库 MySQL 协议）

    参数：
        sql: SQL 语句
        params: 查询参数列表

    返回值：
        (列名列表, 行列表) 元组，每行为 dict
    """
    client = get_starrocks_connection()
    if client is None:
        return [], []
    try:
        return client.query(sql)
    finally:
        client.close()

_TB_FILE = "./data/tb.json"


def _init_file_storage():
    """初始化文件存储目录和示例数据"""
    DATA_DIR = "./data"
    global _TB_FILE
    _TB_FILE = os.path.join(DATA_DIR, "tb.json")
    os.makedirs(DATA_DIR, exist_ok=True)
    if not os.path.exists(_TB_FILE):
        sample = [
            {"a": "TkZEV19ZQV9ORkRXX1lYX3Rlc3RfNS", "b": "示例数据B1", "c": "示例数据C1"},
            {"a": "another_unique_code_2", "b": "示例数据B2", "c": "示例数据C2"},
        ]
        with open(_TB_FILE, "w", encoding="utf-8") as f:
            json.dump(sample, f, ensure_ascii=False, indent=2)


def query_file_ids_from_db(limit: int = 10) -> List[Dict[str, Any]]:
    """
    从数据库或文件查询待处理的文件记录
    
    参数：
        limit: 返回记录数量限制
    
    返回值：
        文件记录列表，每个记录为 dict
    """
    if USE_FILE_STORAGE:
        _init_file_storage()
        with open(_TB_FILE, "r", encoding="utf-8") as f:
            return json.load(f)[:limit]
    else:
        _, rows = execute_query(_QUERY_SQL, [limit])
        return rows


# ---------------------------------------------------------------------------
# 主执行函数（来自原脚本）
# ---------------------------------------------------------------------------

def main(**params):
    """
    主函数：执行文件列表查询
    
    参数（兼容 workflow 传入的 params）：
        input: 输入数据（支持 dict/JSON字符串/JS对象字面量）
    
    返回值：
        包含 files、start_time、query_step 的字典
    """
    input_data = params.get("input", {})
    if isinstance(input_data, dict):
        limit = input_data.get("limit", 10)
    else:
        limit = 10

    start_time = datetime.now().isoformat()

    query_step = {
        "step": 1,
        "name": "执行StarRocks查询",
        "start_time": datetime.now().isoformat(),
        "end_time": None,
        "duration_ms": None,
        "status": "pending",
        "sql": _QUERY_SQL.replace("%s", str(limit)),
        "record_count": 0,
        "error": None,
    }

    t0 = time.time()
    try:
        query_step["status"] = "running"
        file_records = query_file_ids_from_db(limit)
        query_step["status"] = "success"
        query_step["record_count"] = len(file_records)
    except Exception as e:
        query_step["status"] = "failed"
        query_step["error"] = str(e)
        file_records = []

    query_step["duration_ms"] = round((time.time() - t0) * 1000, 2)
    query_step["end_time"] = datetime.now().isoformat()

    return {
        "output": {
            "files": file_records,
            "start_time": start_time,
            "query_step": query_step,
        }
    }

def parse_args():
    """
    解析命令行参数
    
    返回值：
        argparse.Namespace: 包含 limit 和 output 属性的对象
    """
    parser = argparse.ArgumentParser(
        description="步骤1命令行版：从 StarRocks 查询待处理文件列表",
        epilog="使用示例：python step1_get_file_list_cli.py --limit 20 --output my_files.json"
    )
    parser.add_argument(
        "--limit",
        type=int,
        default=10,
        help="查询文件数量限制，默认 10"
    )
    parser.add_argument(
        "--output",
        type=str,
        default="file_list.json",
        help="输出 JSON 文件路径，默认 file_list.json"
    )
    return parser.parse_args()


def cli_main():
    """
    命令行主入口函数
    
    执行流程：
        1. 解析命令行参数
        2. 调用 main() 获取查询结果
        3. 将结果输出到指定的 JSON 文件
    """
    args = parse_args()
    
    print(f"开始查询文件列表，limit={args.limit}")
    
    # 调用主函数，传入 limit 参数
    result = main(input={"limit": args.limit})
    output_data = result["output"]
    
    # 输出到 JSON 文件
    output_path = args.output
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(output_data, f, ensure_ascii=False, indent=2)
    
    print(f"查询完成，共获取 {output_data['query_step']['record_count']} 条记录")
    print(f"结果已保存到: {output_path}")
    
    return output_data


if __name__ == "__main__":
    cli_main()
